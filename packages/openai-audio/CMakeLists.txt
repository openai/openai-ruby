cmake_minimum_required(VERSION 3.16)
project(openai_audio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(PORTAUDIO_INCLUDE_DIR portaudio.h REQUIRED)
find_library(PORTAUDIO_LIBRARY NAMES portaudio REQUIRED)
find_path(SAMPLERATE_INCLUDE_DIR samplerate.h REQUIRED)
find_library(SAMPLERATE_LIBRARY NAMES samplerate REQUIRED)
find_package(Threads REQUIRED)
add_executable(openai-audio-worker native/worker.cpp)
target_include_directories(openai-audio-worker PRIVATE ${PORTAUDIO_INCLUDE_DIR} ${SAMPLERATE_INCLUDE_DIR})
target_link_libraries(openai-audio-worker PRIVATE ${PORTAUDIO_LIBRARY} ${SAMPLERATE_LIBRARY} Threads::Threads)
if(APPLE)
  foreach(framework CoreAudio AudioToolbox AudioUnit CoreFoundation CoreServices)
    find_library(${framework}_LIBRARY ${framework} REQUIRED)
    target_link_libraries(openai-audio-worker PRIVATE ${${framework}_LIBRARY})
  endforeach()
endif()
install(TARGETS openai-audio-worker RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(audio-ring-test test/audio_ring_test.cpp)
  target_link_libraries(audio-ring-test PRIVATE Threads::Threads)
  add_test(NAME audio-ring COMMAND audio-ring-test)
  add_executable(openai-audio-test-worker native/worker.cpp test/fake_portaudio.cpp)
  target_include_directories(openai-audio-test-worker PRIVATE ${PORTAUDIO_INCLUDE_DIR} ${SAMPLERATE_INCLUDE_DIR})
  target_link_libraries(openai-audio-test-worker PRIVATE ${SAMPLERATE_LIBRARY} Threads::Threads)
endif()
